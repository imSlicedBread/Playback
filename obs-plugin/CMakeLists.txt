cmake_minimum_required(VERSION 3.28)
project(fplayback-youtube VERSION 0.1.11 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
option(PLAYBACK_OBS33 "Build for the pinned OBS 33 Chromium 150 runtime" OFF)
add_compile_definitions(PLAYBACK_OBS33=$<BOOL:${PLAYBACK_OBS33}>)
set(PLAYBACK_INSTALLER_SUFFIX "")
if(PLAYBACK_OBS33)
  set(PLAYBACK_INSTALLER_SUFFIX "-OBS33")
endif()
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "This prototype supports Windows x64 only")
endif()
set(OBS_HEADERS "" CACHE PATH "OBS source checkout matching the installed runtime")
set(OBS_IMPORT_LIBS "" CACHE PATH "Import libraries generated from the installed OBS DLLs")
set(OBS_BROWSER_HEADERS "" CACHE PATH "Pinned obs-browser panel header directory")
find_package(Qt6 6.11 REQUIRED COMPONENTS Core Widgets Network)
set(CEF_ROOT "" CACHE PATH "Pinned OBS CEF 127.145.7 SDK")
set(CEF_RUNTIME_LIBRARY_FLAG "/MD" CACHE STRING "Match Qt dynamic runtime" FORCE)
set(USE_SANDBOX OFF CACHE BOOL "Reuse OBS's existing unsandboxed runtime" FORCE)
list(APPEND CMAKE_MODULE_PATH "${CEF_ROOT}/cmake")
find_package(CEF REQUIRED)
add_subdirectory("${CEF_ROOT}/libcef_dll" "cef-wrapper" EXCLUDE_FROM_ALL)
set_target_properties(libcef_dll_wrapper PROPERTIES AUTOMOC OFF MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")
add_library(fplayback-youtube MODULE src/plugin.cpp src/capture.cpp src/audio.cpp src/scene-setup.cpp src/filtered-browser.cpp resources.qrc)
target_include_directories(fplayback-youtube PRIVATE
  "${OBS_HEADERS}/libobs" "${OBS_HEADERS}/libobs-winrt" "${OBS_HEADERS}/frontend/api" "${OBS_IMPORT_LIBS}" "${OBS_BROWSER_HEADERS}" "${CEF_ROOT}")
target_link_libraries(fplayback-youtube PRIVATE Qt6::Core Qt6::Widgets Qt6::Network
  "${OBS_IMPORT_LIBS}/obs.lib" "${OBS_IMPORT_LIBS}/obs-frontend-api.lib" "${OBS_IMPORT_LIBS}/libobs-winrt.lib"
  "${CEF_ROOT}/Release/libcef.lib" libcef_dll_wrapper delayimp Mmdevapi Ole32 Psapi)
target_link_options(fplayback-youtube PRIVATE "/DELAYLOAD:libcef.dll")
target_compile_definitions(fplayback-youtube PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
install(TARGETS fplayback-youtube LIBRARY DESTINATION obs-plugins/64bit RUNTIME DESTINATION obs-plugins/64bit)
enable_testing()
add_executable(playback-ad-response-test src/ad-response-test.cpp)
set_target_properties(playback-ad-response-test PROPERTIES AUTOMOC OFF)
target_link_libraries(playback-ad-response-test PRIVATE Qt6::Core)
add_test(NAME playback-ad-response COMMAND playback-ad-response-test)
set_tests_properties(playback-ad-response PROPERTIES ENVIRONMENT "PATH=${OBS_RUNTIME_PATH}\\;$ENV{PATH}")
add_executable(playback-filtered-browser-test src/filtered-browser-test.cpp src/filtered-browser.cpp)
target_include_directories(playback-filtered-browser-test PRIVATE "${CEF_ROOT}")
target_compile_definitions(playback-filtered-browser-test PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(playback-filtered-browser-test PRIVATE Qt6::Widgets Qt6::Network "${CEF_ROOT}/Release/libcef.lib" libcef_dll_wrapper delayimp Psapi)
target_link_options(playback-filtered-browser-test PRIVATE "/DELAYLOAD:libcef.dll")
find_package(Python3 3.10 REQUIRED COMPONENTS Interpreter)
add_test(NAME playback-filtered-browser COMMAND playback-filtered-browser-test "${OBS_RUNTIME_PATH}/../.." "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/src/https-fixture-server.py")
set_tests_properties(playback-filtered-browser PROPERTIES TIMEOUT 25 ENVIRONMENT "PATH=${OBS_RUNTIME_PATH}\\;$ENV{PATH};QT_QPA_PLATFORM_PLUGIN_PATH=${CMAKE_PREFIX_PATH}/plugins/platforms")
add_test(NAME playback-browser-soak COMMAND playback-filtered-browser-test "${OBS_RUNTIME_PATH}/../.." "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/src/https-fixture-server.py" --soak)
set_tests_properties(playback-browser-soak PROPERTIES TIMEOUT 180 RUN_SERIAL TRUE ENVIRONMENT "PATH=${OBS_RUNTIME_PATH}\\;$ENV{PATH};QT_QPA_PLATFORM_PLUGIN_PATH=${CMAKE_PREFIX_PATH}/plugins/platforms")
add_executable(playback-ad-output-guard-test src/ad-output-guard-test.cpp)
set_target_properties(playback-ad-output-guard-test PROPERTIES AUTOMOC OFF)
add_test(NAME playback-ad-output-guard COMMAND playback-ad-output-guard-test)
add_executable(playback-recovery-policy-test src/recovery-policy-test.cpp)
set_target_properties(playback-recovery-policy-test PROPERTIES AUTOMOC OFF)
add_test(NAME playback-recovery-policy COMMAND playback-recovery-policy-test)
add_executable(playback-output-resolution-test src/output-resolution-test.cpp)
set_target_properties(playback-output-resolution-test PROPERTIES AUTOMOC OFF)
add_test(NAME playback-output-resolution COMMAND playback-output-resolution-test)
add_executable(playback-video-fade-test src/video-fade-test.cpp)
set_target_properties(playback-video-fade-test PROPERTIES AUTOMOC OFF)
add_test(NAME playback-video-fade COMMAND playback-video-fade-test)
add_executable(playback-video-fade-gpu-test src/video-fade-gpu-test.cpp)
set_target_properties(playback-video-fade-gpu-test PROPERTIES AUTOMOC OFF)
target_include_directories(playback-video-fade-gpu-test PRIVATE "${OBS_HEADERS}/libobs" "${OBS_IMPORT_LIBS}")
target_link_libraries(playback-video-fade-gpu-test PRIVATE "${OBS_IMPORT_LIBS}/obs.lib")
add_test(NAME playback-video-fade-gpu COMMAND playback-video-fade-gpu-test "${OBS_RUNTIME_PATH}/libobs-d3d11.dll")
set_tests_properties(playback-video-fade-gpu PROPERTIES WORKING_DIRECTORY "${OBS_RUNTIME_PATH}")

add_executable(playback-scene-setup-test src/scene-setup-test.cpp src/scene-setup.cpp)
set_target_properties(playback-scene-setup-test PROPERTIES AUTOMOC OFF)
target_include_directories(playback-scene-setup-test PRIVATE "${OBS_HEADERS}/libobs" "${OBS_IMPORT_LIBS}")
target_link_libraries(playback-scene-setup-test PRIVATE "${OBS_IMPORT_LIBS}/obs.lib")
add_test(NAME playback-scene-setup COMMAND playback-scene-setup-test)
set_tests_properties(playback-scene-setup PROPERTIES WORKING_DIRECTORY "${OBS_RUNTIME_PATH}")

enable_language(RC)
file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/installer-$<CONFIG>.rc" CONTENT
"101 RCDATA \"$<TARGET_FILE:fplayback-youtube>\"\n102 RCDATA \"${INSTALLER_SOURCE_ZIP}\"\n103 RCDATA \"${CMAKE_CURRENT_SOURCE_DIR}/LICENSE\"\n1 24 \"${CMAKE_CURRENT_SOURCE_DIR}/src/installer.manifest\"\n")
add_executable(fplayback-plugin-setup WIN32 src/installer.cpp "${CMAKE_CURRENT_BINARY_DIR}/installer-$<CONFIG>.rc")
set_target_properties(fplayback-plugin-setup PROPERTIES AUTOMOC OFF OUTPUT_NAME "fKey-Playback-OBS-Plugin-Setup-${PROJECT_VERSION}${PLAYBACK_INSTALLER_SUFFIX}"
  MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
target_link_libraries(fplayback-plugin-setup PRIVATE Shell32 Ole32 Advapi32 Version User32 Gdi32)
target_link_options(fplayback-plugin-setup PRIVATE /MANIFEST:NO)
add_dependencies(fplayback-plugin-setup fplayback-youtube)
install(TARGETS fplayback-plugin-setup RUNTIME DESTINATION .)

add_executable(playback-installer-test src/installer-test.cpp)
set_target_properties(playback-installer-test PROPERTIES AUTOMOC OFF)
target_link_libraries(playback-installer-test PRIVATE Version)
add_dependencies(playback-installer-test fplayback-plugin-setup)
add_test(NAME playback-plugin-installer COMMAND playback-installer-test "$<TARGET_FILE:fplayback-plugin-setup>"
  "${OBS_RUNTIME_PATH}/../.." "$<TARGET_FILE:fplayback-youtube>" "${INSTALLER_SOURCE_ZIP}")
